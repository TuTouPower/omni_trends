```ts
import i18n from "i18next"
import { initReactI18next } from "react-i18next"
import zh from "./locales/zh.json"
import en from "./locales/en.json"

function detectLang(): "zh" | "en" {
  try {
    return navigator.language.startsWith("zh") ? "zh" : "en"
  } catch {
    return "zh"
  }
}

const savedLang = (() => {
  try {
    return localStorage.getItem("lang") as "zh" | "en" | null
  } catch {
    return null
  }
})()

i18n.use(initReactI18next).init({
  resources: {
    zh: { translation: zh },
    en: { translation: en },
  },
  lng: savedLang || detectLang(),
  fallbackLng: "zh",
  interpolation: { escapeValue: false },
})

export { detectLang }
export default i18n
```

- [ ] **Step 3: 创建 `src/hooks/useLang.ts`**

```ts
import i18n, { detectLang } from "~/i18n"

export type Lang = "zh" | "en"

const langAtom = atomWithStorage<Lang>("lang", detectLang())

export function useLang() {
  const [lang, setLang_] = useAtom(langAtom)

  const setLang = useCallback((l: Lang) => {
    setLang_(l)
    i18n.changeLanguage(l)
  }, [setLang_])

  return { lang, setLang }
}
```

注：`atom`, `useAtom`, `atomWithStorage`, `useCallback` 均由 unimport 自动导入。

```ts
  if (!timestamp) return undefined
  const date = new Date(timestamp)

  const now = new Date()
```

shared 层不能 import `~/i18n`。改方案：`relativeTime` 保持原样返回中文，在前端组件中用 i18next key 替代。

**实际方案：** `relativeTime` 保持不变。在使用它的组件中，改为用 i18next key `time.*` 直接计算，不依赖 `relativeTime` 函数。

在 `src/hooks/useRelativeTime.ts` 中新增一个 `useRelativeTimeI18n` 函数（或者直接修改现有函数支持 i18n）：

```ts
// src/hooks/useRelativeTime.ts — 替换现有实现
import i18n from "~/i18n"

export function relativeTimeI18n(timestamp: string | number): string | undefined {
  if (!timestamp) return undefined
  const date = new Date(timestamp)
  if (Number.isNaN(date.getDay())) return undefined

  const now = new Date()
  const diffInSeconds = (now.getTime() - date.getTime()) / 1000
  const diffInMinutes = diffInSeconds / 60
  const diffInHours = diffInMinutes / 60

  const t = i18n.t.bind(i18n)

  if (diffInSeconds < 60) {
    return t("time.justNow")
  } else if (diffInMinutes < 60) {
    return t("time.minutesAgo", { count: Math.floor(diffInMinutes) })
  } else if (diffInHours < 24) {
    return t("time.hoursAgo", { count: Math.floor(diffInHours) })
  } else {
    return t("time.date", { month: date.getMonth() + 1, day: date.getDate() })
  }
}

// 保留旧的 relativeTime 调用兼容
export function useRelativeTime(timestamp: string | number) {
  const [time, setTime] = useState<string>()
  const timer = useAtomValue(timerAtom)
  const visible = useVisibility()

  useEffect(() => {
    if (visible) {
      const t = relativeTimeI18n(timestamp)
      if (t) setTime(t)
    }
  }, [timestamp, timer, visible])

  return time
}
```

注：`timerAtom`, `useVisibility` 保持原样不动。

修改 `groupByColumn` 函数（在 SearchBar 文件内），改为用 column key 排序：

```tsx
interface SourceItemProps {
  id: SourceID
  name: string
  title?: string
  column: string | null
  columnName: string
  pinyin: string
}

function groupByColumn(items: SourceItemProps[]) {
  return items.reduce((acc, item) => {
    const k = acc.find(i => i.column === item.column)
    if (k) k.sources = [...k.sources, item]
    else acc.push({ column: item.column, columnName: item.columnName, sources: [item] })
    return acc
  }, [] as {
    column: string | null
    columnName: string
    sources: SourceItemProps[]
  }[]).sort((m, n) => {
    if (m.column === "tech") return -1
    if (n.column === "tech") return 1
    if (!m.column) return 1
    if (!n.column) return -1
    return m.column < n.column ? -1 : 1
  })
}
```
